# fw/: the platform firmware.
#
# acpi/: the ACPI tables (the DSDT in ASL, compiled by iasl; the rest built
# by mktables.py) and QEMU's table-loader script; the firmware links them
# into its flash image and vsim can also serve them through fw_cfg.
#
# The UEFI firmware (fw/README.md): SEC (assembly), PEI and the DXE images
# (C, PE32+ for x86_64-unknown-windows, linked with lld-link), packed into
# build/fw/vibe86.fd by tools/mkfd.py.

# ---------------------------------------------------------------- UEFI images

set(FW_SRC ${CMAKE_CURRENT_SOURCE_DIR})
set(FW_OUT ${CMAKE_CURRENT_BINARY_DIR})
# the platform header (vibe86_platform.h), generated by rtl/common at configure time
set(FW_PLATFORM_INC ${VIBE86_COMMON_DIR})

# No SSE/x87 yet on v1 (fw/README.md): general registers only. The Microsoft
# x64 convention is UEFI's (UEFI 2.10 2.3.4); no stack probes, no red zone
# (interrupt handlers share the stack).
set(FW_CFLAGS -target x86_64-unknown-windows -std=c11 -O2 -ffreestanding -fno-builtin
    -mgeneral-regs-only -mno-red-zone -mno-stack-arg-probe -fno-stack-protector
    -fshort-wchar -fno-strict-aliasing -fno-asynchronous-unwind-tables -fno-unwind-tables
    -Wall -Wextra -Wno-unused-parameter -Werror
    -nostdinc -I${FW_SRC}/include/std -I${FW_SRC}/include -I${FW_PLATFORM_INC}
    -I${CMAKE_SOURCE_DIR}/sim)	# sim/vgafont.h, the VGA font (VideoDxe)

set(FW_ASFLAGS -target x86_64-unknown-windows -nostdinc -I${FW_SRC}/include)

set(FW_LIB_SOURCES lib/string.c lib/print.c lib/io.c lib/devpath.c lib/pe.c lib/fv.c)

# fw_objects(<out var> <name> <sources...>): compile with FW_CFLAGS
function(fw_objects out name)
  set(objs)
  foreach(src IN LISTS ARGN)
    get_filename_component(base ${src} NAME_WE)
    get_filename_component(dir ${src} DIRECTORY)
    string(REPLACE "/" "_" dir "${dir}")
    set(obj ${FW_OUT}/obj/${name}/${dir}_${base}.obj)
    if(src MATCHES "\\.S$")
      set(flags ${FW_ASFLAGS})
    else()
      set(flags ${FW_CFLAGS})
    endif()
    add_custom_command(OUTPUT ${obj}
      COMMAND ${CMAKE_COMMAND} -E make_directory ${FW_OUT}/obj/${name}
      COMMAND ${FW_CLANG} ${flags} -MD -MF ${obj}.d -c -o ${obj} ${FW_SRC}/${src}
      DEPENDS ${FW_SRC}/${src}
      DEPFILE ${obj}.d
      COMMENT "fw ${name}: ${src}"
      VERBATIM)
    list(APPEND objs ${obj})
  endforeach()
  set(${out} ${objs} PARENT_SCOPE)
endfunction()

# fw_image(<name> SUBSYSTEM <efi_...> [ENTRY sym] [LINK flags...] SOURCES ...)
function(fw_image name)
  cmake_parse_arguments(A "" "SUBSYSTEM;ENTRY" "SOURCES;LINK" ${ARGN})
  if(NOT A_ENTRY)
    set(A_ENTRY efi_main)
  endif()
  fw_objects(objs ${name} ${A_SOURCES} ${FW_LIB_SOURCES})
  set(img ${FW_OUT}/${name}.efi)
  add_custom_command(OUTPUT ${img}
    COMMAND ${FW_LLD_LINK} /nologo /nodefaultlib /subsystem:${A_SUBSYSTEM} /entry:${A_ENTRY}
            /machine:x64 /debug:none /merge:.rdata=.text ${A_LINK} /out:${img} ${objs}
    DEPENDS ${objs}
    COMMENT "fw ${name}.efi"
    VERBATIM)
  add_custom_target(fw-${name} DEPENDS ${img})
  set(FW_IMAGE_${name} ${img} PARENT_SCOPE)
endfunction()

# fw_driver(<name> [RUNTIME] SOURCES ...): a DXE driver, dispatched in the
# order of the calls; fw_app(<name> SOURCES ...): an application
set(FW_DRIVERS)
function(fw_driver name)
  cmake_parse_arguments(A "RUNTIME" "" "SOURCES" ${ARGN})
  if(A_RUNTIME)
    set(sub efi_runtime_driver)
  else()
    set(sub efi_boot_service_driver)
  endif()
  fw_image(${name} SUBSYSTEM ${sub} SOURCES ${A_SOURCES} ${FW_UEFILIB_SOURCES})
  set(FW_DRIVERS ${FW_DRIVERS} ${name} PARENT_SCOPE)
  set(FW_IMAGE_${name} ${FW_IMAGE_${name}} PARENT_SCOPE)
endfunction()
