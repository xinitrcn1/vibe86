# fw/: the platform firmware.
#
# acpi/: the ACPI tables (the DSDT in ASL, compiled by iasl; the rest built
# by mktables.py) and QEMU's table-loader script; the firmware links them
# into its flash image and vsim can also serve them through fw_cfg.
#
# The UEFI firmware (fw/README.md): SEC (assembly), PEI and the DXE images
# (C, PE32+ for x86_64-unknown-windows, linked with lld-link), packed into
# build/fw/vibe86.fd by tools/mkfd.py.

find_program(IASL iasl REQUIRED)
find_program(FW_CLANG clang REQUIRED)
find_program(FW_LLD_LINK NAMES lld-link lld-link19 lld-link18 REQUIRED)
find_program(FW_LD_LLD ld.lld REQUIRED)

set(ACPI_OUT ${CMAKE_CURRENT_BINARY_DIR}/acpi)
add_custom_command(
  OUTPUT ${ACPI_OUT}/dsdt.aml ${ACPI_OUT}/tables ${ACPI_OUT}/rsdp ${ACPI_OUT}/table-loader
  COMMAND ${CMAKE_COMMAND} -E make_directory ${ACPI_OUT}
  COMMAND ${IASL} -vs -p ${ACPI_OUT}/dsdt ${CMAKE_CURRENT_SOURCE_DIR}/acpi/dsdt.asl
  COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/acpi/mktables.py
          ${ACPI_OUT}/dsdt.aml ${ACPI_OUT} --platform ${VIBE86_PLATFORM_TOML}
  DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/acpi/dsdt.asl ${CMAKE_CURRENT_SOURCE_DIR}/acpi/mktables.py
          ${VIBE86_PLATFORM_TOML}
  COMMENT "ACPI tables (iasl, mktables.py)"
  VERBATIM)
add_custom_target(acpi ALL DEPENDS ${ACPI_OUT}/tables)
set(VIBE86_ACPI_DIR ${ACPI_OUT} CACHE INTERNAL "")

# ---------------------------------------------------------------- UEFI images

set(FW_SRC ${CMAKE_CURRENT_SOURCE_DIR})
set(FW_OUT ${CMAKE_CURRENT_BINARY_DIR})
# the platform header (vibe86_platform.h), generated by rtl/common at configure time
set(FW_PLATFORM_INC ${VIBE86_COMMON_DIR})

# No SSE/x87 yet on v1 (fw/README.md): general registers only. The Microsoft
# x64 convention is UEFI's (UEFI 2.10 2.3.4); no stack probes, no red zone
# (interrupt handlers share the stack).
set(FW_CFLAGS -target x86_64-unknown-windows -std=c11 -O2 -ffreestanding -fno-builtin
    -mgeneral-regs-only -mno-red-zone -mno-stack-arg-probe -fno-stack-protector
    -fshort-wchar -fno-strict-aliasing -fno-asynchronous-unwind-tables -fno-unwind-tables
    -Wall -Wextra -Wno-unused-parameter -Werror
    -nostdinc -I${FW_SRC}/include/std -I${FW_SRC}/include -I${FW_PLATFORM_INC})

set(FW_LIB_SOURCES lib/string.c lib/print.c lib/io.c lib/devpath.c lib/pe.c lib/fv.c)

# fw_objects(<out var> <name> <sources...>): compile with FW_CFLAGS
function(fw_objects out name)
  set(objs)
  foreach(src IN LISTS ARGN)
    get_filename_component(base ${src} NAME_WE)
    get_filename_component(dir ${src} DIRECTORY)
    string(REPLACE "/" "_" dir "${dir}")
    set(obj ${FW_OUT}/obj/${name}/${dir}_${base}.obj)
    add_custom_command(OUTPUT ${obj}
      COMMAND ${CMAKE_COMMAND} -E make_directory ${FW_OUT}/obj/${name}
      COMMAND ${FW_CLANG} ${FW_CFLAGS} -MD -MF ${obj}.d -c -o ${obj} ${FW_SRC}/${src}
      DEPENDS ${FW_SRC}/${src}
      DEPFILE ${obj}.d
      COMMENT "fw ${name}: ${src}"
      VERBATIM)
    list(APPEND objs ${obj})
  endforeach()
  set(${out} ${objs} PARENT_SCOPE)
endfunction()

# fw_image(<name> SUBSYSTEM <efi_...> [ENTRY sym] [LINK flags...] SOURCES ...)
function(fw_image name)
  cmake_parse_arguments(A "" "SUBSYSTEM;ENTRY" "SOURCES;LINK" ${ARGN})
  if(NOT A_ENTRY)
    set(A_ENTRY efi_main)
  endif()
  fw_objects(objs ${name} ${A_SOURCES} ${FW_LIB_SOURCES})
  set(img ${FW_OUT}/${name}.efi)
  add_custom_command(OUTPUT ${img}
    COMMAND ${FW_LLD_LINK} /nologo /nodefaultlib /subsystem:${A_SUBSYSTEM} /entry:${A_ENTRY}
            /machine:x64 /debug:none /merge:.rdata=.text ${A_LINK} /out:${img} ${objs}
    DEPENDS ${objs}
    COMMENT "fw ${name}.efi"
    VERBATIM)
  add_custom_target(fw-${name} DEPENDS ${img})
  set(FW_IMAGE_${name} ${img} PARENT_SCOPE)
endfunction()

# SEC: flat 64 KiB at the top of the flash
set(SEC_OBJ ${FW_OUT}/obj/sec.o)
set(SEC_BIN ${FW_OUT}/sec.bin)
add_custom_command(OUTPUT ${SEC_BIN}
  COMMAND ${CMAKE_COMMAND} -E make_directory ${FW_OUT}/obj
  COMMAND ${FW_CLANG} -c -target x86_64-unknown-none -I${FW_SRC}/include -o ${SEC_OBJ} ${FW_SRC}/sec/sec.S
  COMMAND ${FW_LD_LLD} -T ${CMAKE_SOURCE_DIR}/tests/flat.ld -o ${SEC_BIN} ${SEC_OBJ}
  DEPENDS ${FW_SRC}/sec/sec.S ${FW_SRC}/include/layout.h
  COMMENT "fw SEC"
  VERBATIM)

# PEI: a fixed-address image run in place (file layout = memory layout)
fw_image(Pei SUBSYSTEM efi_boot_service_driver ENTRY pei_main
  LINK /base:0xFFF00000 /fixed /filealign:4096
  SOURCES pei/pei.c)

fw_image(DxeCore SUBSYSTEM efi_boot_service_driver ENTRY dxe_main
  SOURCES dxe/core/main.c)

set(FD ${FW_OUT}/vibe86.fd)
add_custom_command(OUTPUT ${FD}
  COMMAND ${Python3_EXECUTABLE} -B ${FW_SRC}/tools/mkfd.py -o ${FD}
          --layout ${FW_SRC}/include/layout.h --sec ${SEC_BIN} --pei ${FW_IMAGE_Pei}
          --core ${FW_IMAGE_DxeCore}
          --raw d6a2cb7f-6a18-4e2f-b43c-766238360010=${ACPI_OUT}/tables
          --raw d6a2cb7f-6a18-4e2f-b43c-766238360011=${ACPI_OUT}/rsdp
          --raw d6a2cb7f-6a18-4e2f-b43c-766238360012=${ACPI_OUT}/table-loader
  DEPENDS ${FW_SRC}/tools/mkfd.py ${FW_SRC}/include/layout.h ${SEC_BIN} ${FW_IMAGE_Pei}
          ${FW_IMAGE_DxeCore} ${ACPI_OUT}/tables
  COMMENT "fw vibe86.fd"
  VERBATIM)
add_custom_target(firmware ALL DEPENDS ${FD})
set(VIBE86_FD ${FD} CACHE INTERNAL "")
